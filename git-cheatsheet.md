git init	start a git repository in this folder
git add	pick which files to include in the next commit
git commit -m	save a save-point with a message
git log --oneline	show the commit history, one line each
git diff	show changes since the last commit
git status	show what's changed, staged, or new
git branch	list branches; * = where you are
git checkout -b	create a new branch and switch to it
git merge	bring another branch's changes into this one
git push	send commits up to GitHub
git pull	bring GitHub's latest commits down to you
git clone	copy a full project from GitHub to your computer
.gitignore	tells git which files to never track
